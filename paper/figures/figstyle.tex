% \stagebox{name}{x,y}{colour}{light colour}{width}{height}{Title}{icons}{text}: a box with a coloured
% title band, an icon row and one short line of text
\newcommand{\stagebox}[9]{%
  \node[boxbody={#3}{#4}, minimum width=#5, minimum height=#6] (#1) at (#2) {};
  \begin{scope}
    \clip[rounded corners=5pt] (#1.south west) rectangle (#1.north east);
    \fill[#3] ([yshift=-0.62cm]#1.north west) rectangle (#1.north east);
  \end{scope}
  \node[title] at ([yshift=-0.31cm]#1.north) {#7};
  \node[inner sep=0pt] (#1-icons) at ([yshift=-1.25cm]#1.north) {#8};
  \node[body, anchor=south, text width=\dimexpr#5-0.3cm\relax] (#1-text) at ([yshift=0.22cm]#1.south) {#9};
}
% \stageboxi{name}{x,y}{colour}{light colour}{width}{height}{Title}{icon}{text}: the same box with the
% icon on the left of the body and the text beside it (for short, wide boxes)
\newcommand{\stageboxi}[9]{%
  \node[boxbody={#3}{#4}, minimum width=#5, minimum height=#6] (#1) at (#2) {};
  \begin{scope}
    \clip[rounded corners=5pt] (#1.south west) rectangle (#1.north east);
    \fill[#3] ([yshift=-0.62cm]#1.north west) rectangle (#1.north east);
  \end{scope}
  \node[title] at ([yshift=-0.31cm]#1.north) {#7};
  \coordinate (#1-mid) at ($(#1.south)!0.5!([yshift=-0.62cm]#1.north)$);
  \node[inner sep=0pt, anchor=west] (#1-icons) at ([xshift=0.32cm]#1.west |- #1-mid) {#8};
  \node[body, anchor=west, align=left, text width=\dimexpr#5-1.55cm\relax] (#1-text) at ([xshift=0.22cm]#1-icons.east) {#9};
}
