% Figure 2: the threat model on the path of a request (manuscript Section 3.2): the attacker's two goals,
% what it controls at each point, and the limits of the model; the details are in the text.

\documentclass[tikz,border=3pt]{standalone}
\input{figstyle}
\begin{document}
\begin{tikzpicture}
\def\w{3.6cm}\def\h{2.35cm}\def\dx{4.55}
\stagebox{client}{0,0}{cClient}{cClientL}{\w}{\h}{Client}
  {\ic{0.58cm}{user-secret-red}\hspace{6pt}\ic{0.52cm}{mask-red}\hspace{6pt}\ic{0.55cm}{laptop-red}}
  {scripts behind proxies}
\stagebox{service}{\dx,0}{cService}{cServiceL}{\w}{\h}{Service Endpoint}
  {\ic{0.58cm}{server-indigo}\hspace{6pt}\ic{0.55cm}{shield-halved-indigo}}
  {the system under study}
\stagebox{vendor}{2*\dx,0}{cVendor}{cVendorL}{\w}{\h}{SMS Vendor}
  {\ic{0.58cm}{tower-broadcast-teal}\hspace{6pt}\ic{0.55cm}{comment-sms-teal}}
  {beyond the attacker's reach}
\stagebox{operator}{3*\dx,0}{cOperator}{cOperatorL}{\w}{\h}{Terminating Operator}
  {\ic{0.58cm}{tower-cell-green}\hspace{6pt}\ic{0.55cm}{coins-green}}
  {the pumper's partner}
% forward path
\draw[fwd=cClient] (client.east) -- node[edgelab, above=1pt]{request} (service.west);
\draw[fwd=cService] (service.east) -- node[edgelab, above=1pt]{send} (vendor.west);
\draw[fwd=cOperator] (vendor.east) -- node[edgelab, above=1pt]{deliver} (operator.west);
\draw[loop=cOperator] (operator.north) -- ++(0,0.45) -| node[edgelab, pos=0.25, above=1pt]{share of the termination fee\hspace{3pt}\raisebox{-2pt}{\ic{0.36cm}{hand-holding-dollar-green}}} (client.north);
% goals, above the loop
\node[chip={cClient}{cClientL}, anchor=south, text width=6.2cm] (goal1) at ($(client.north)!0.5!(service.north)+(0,1.25)$)
  {\raisebox{-2pt}{\ic{0.34cm}{bolt-red}}\hspace{4pt}\textbf{Flooder:} make the service pay for messages nobody asked for};
\node[chip={cOperator}{cOperatorL}, anchor=south, text width=6.2cm] (goal2) at ($(vendor.north)!0.5!(operator.north)+(0,1.25)$)
  {\raisebox{-2pt}{\ic{0.34cm}{sack-dollar-green}}\hspace{4pt}\textbf{Pumper:} be paid a share of the fee on its partner's ranges};
% what the attacker controls at each point, below the path
\node[chip={cClient}{cClientL}, below=0.5cm of client, text width=3.3cm] (c1)
  {\textbf{controls here}\\proxies, bought CAPTCHA tokens, fresh fingerprints, the numbers it asks for};
\node[chip={cService}{cServiceL}, below=0.5cm of service, text width=3.3cm] (c2)
  {\textbf{knows}\\every threshold and timing; sees one uniform response};
\node[chip={cVendor}{cVendorL}, below=0.5cm of vendor, text width=3.3cm] (c3)
  {\textbf{cannot touch}\\the route, the receipts or the service's state};
\node[chip={cOperator}{cOperatorL}, below=0.5cm of operator, text width=3.3cm] (c4)
  {\textbf{controls on its own ranges}\\enters codes, fakes receipts, builds trust before flooding};
% limits of the model
\node[chip={cSlate}{cSlateL}, text width=16.6cm, anchor=north] at ($(c2.south)!0.5!(c3.south)+(0,-0.35)$)
  {\raisebox{-2pt}{\ic{0.32cm}{ban-slate}}\hspace{4pt}\textbf{cannot} forge an attestation, earn on ranges its partner does not terminate, or make a stranger enter a code\\\textbf{outside the model:} attacks on a delivered code; compromise of the service, its vendors or the attestation providers};
\end{tikzpicture}
\end{document}
