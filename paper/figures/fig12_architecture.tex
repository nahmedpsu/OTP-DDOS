\documentclass[tikz,border=2pt]{standalone}
\usepackage{times}
\usetikzlibrary{positioning,arrows.meta,calc,fit,backgrounds}
\begin{document}
\begin{tikzpicture}[
  comp/.style={draw, rounded corners=2pt, align=left, font=\scriptsize, inner sep=3.5pt, fill=white},
  party/.style={draw, rounded corners=2pt, align=center, font=\scriptsize, inner sep=3.5pt, fill=gray!10},
  ext/.style={draw, align=left, font=\scriptsize, inner sep=3pt, fill=yellow!8, text width=3.5cm},
  store/.style={draw, rounded corners=2pt, align=left, font=\scriptsize, inner sep=3.5pt, fill=blue!5},
  arr/.style={-{Latex[length=1.8mm]}, thick},
  ret/.style={-{Latex[length=1.8mm]}, thick, dashed},
  both/.style={{Latex[length=1.6mm]}-{Latex[length=1.6mm]}, thick},
  lab/.style={font=\tiny, fill=white, inner sep=1pt, align=center},
  title/.style={font=\scriptsize\bfseries}
]
% ---- the service ----
\node[comp, text width=8.5cm] (api) at (7.8,0) {\textbf{HTTP API} (FastAPI; any number of instances)\\
public: \texttt{/attest/challenge}, \texttt{/attest/enroll}, \texttt{/session} (a reCAPTCHA token or an attestation buys a signed session), \texttt{/otp/request} (the pipeline), \texttt{/otp/verify} (the code)\\
internal, behind a service credential: \texttt{/internal/delivery} (the vendor's receipt), \texttt{/internal/timeouts/run}};

\node[comp, text width=3.95cm, anchor=north east] (pipe) at ($(api.south east)+(0,-0.3)$) {\textbf{Pipeline} (Steps 0--11, \textsc{ProcessRequest})\\
cheap checks first; every limit one atomic store operation, released on refusal; reserve, write state, enqueue; one uniform body after 400\,ms};
\node[comp, text width=3.95cm, anchor=north west] (fb) at ($(api.south west)+(0,-0.3)$) {\textbf{Feedback loop} (\textsc{OnOutcome})\\
one compare-and-set per transition of a send (delivered, verified, failed, undelivered); reputation per key; block statistics, verdicts; denylist};

\node[store, text width=8.5cm, anchor=north] (st) at ($(api.south)+(0,-2.5)$) {\textbf{Store} (Redis in production; an in-memory twin with the same rules for the tests and the simulator)\\
signed sessions and nonces; per-key limits and reservations; number tracker; SMS history, one entry per send; reputation counts per key; block statistics, counters and verdicts; denylist; adaptive multipliers; operating mode, hourly count and spend. What the loop writes, the next request reads.};

\node[comp, text width=8.5cm, anchor=north] (worker) at ($(st.south)+(0,-0.45)$) {\textbf{Worker} (one per deployment; a one-minute tick)\\
resolves the sends that timed out (the feedback loop's undelivered and failed transitions); runs the baseline job: the expected rate per source, platform and country for this hour of the week, from the pipeline's own counters, sets the adaptive multipliers of Step~9};

\begin{scope}[on background layer]
\node[draw, rounded corners=3pt, fill=gray!4, inner sep=6pt, fit=(api)(pipe)(fb)(st)(worker)] (svc) {};
\end{scope}
\node[title, anchor=south west] at (svc.north west) {The service (release 2.9.0)};

% internal edges
\draw[both] (pipe.south) -- node[lab, right]{read; atomic take and reserve} (pipe.south |- st.north);
\draw[arr] (fb.south) -- node[lab, right]{compare-and-set} (fb.south |- st.north);
\draw[both] (worker.north) -- node[lab, right]{timeouts; counters $\to$ multipliers} (worker.north |- st.south);

% ---- client and recipient ----
\node[party, text width=2.2cm] (client) at (0.5,0.05) {\textbf{Client}\\web or app};
\node[party, text width=2.2cm] (recip) at (0.5,-1.85) {\textbf{Recipient}\\the number's owner};
\draw[arr] ([yshift=0.3cm]client.east) -- node[lab, above]{session,\\request} ([yshift=0.3cm]client.east -| svc.west);
\draw[arr] ([yshift=-0.3cm]client.east -| svc.west) -- node[lab, below]{uniform body,\\or a challenge} ([yshift=-0.3cm]client.east);
\draw[ret] (recip.east) -- node[lab, above]{the code, or nothing} (recip.east -| svc.west);

% ---- external services ----
\node[ext, anchor=north west] (sms) at (12.95,0.6) {\textbf{SMS vendor} (Step 11)\\sends the message; reports a delivery receipt};
\node[ext, below=0.15cm of sms] (push) {\textbf{Push, WhatsApp}\\the downgrade channels};
\node[ext, below=0.15cm of push] (att) {\textbf{App Attest, Play Integrity}\\Step 0: the platform};
\node[ext, below=0.15cm of att] (rc) {\textbf{reCAPTCHA v3; the challenge}\\Steps 3 and 7, web only};
\node[ext, below=0.15cm of rc] (ip) {\textbf{Address intelligence, proxy lists}\\Steps 0 and 2};
\node[ext, below=0.15cm of ip] (hlr) {\textbf{HLR lookup; prefix table}\\Step 5: liveness, cost class};
\node[ext, below=0.15cm of hlr] (al) {\textbf{Alert sink}\\circuit breaker, outages};
\node[title, anchor=south west, align=left, text width=3.6cm] at ([yshift=0.05cm]sms.north west) {External services, each behind an adapter with an in-memory fake};

\draw[arr] ([yshift=0.15cm]sms.west -| svc.east) -- ([yshift=0.15cm]sms.west);
\draw[ret] ([yshift=-0.2cm]sms.west) -- node[lab, below]{receipt} ([yshift=-0.2cm]sms.west -| svc.east);
\foreach \n in {push, att, rc, ip, hlr, al} {\draw[arr] (\n.west -| svc.east) -- (\n.west);}
\end{tikzpicture}
