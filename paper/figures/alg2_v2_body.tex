% The v2 pipeline (Section III-C) and its feedback loop (Section III-D), as implemented in src/otp_guard/pipeline.py
% and feedback.py (release 2.9.0). Shared by main.tex (Algorithm 2) and the standalone alg2_v2.tex for Word.
% ProcessRequest follows the steps of Figure 4 in order; the symbols, limits and constants are explained in Section III-C.
\begin{algorithmic}[1]
\Require request $r$; the store $S$
\Ensure one uniform reply, after a send, a downgrade, a delay or a refusal
\Procedure{ProcessRequest}{$r$}
\State $p \gets \textsc{Attest}(r)$;\ \textbf{if} $p = \bot$ \textbf{or} $r.\mathrm{address} \in \mathcal{V}$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $\neg\textsc{ValidSession}(r)$ \textbf{or} $n_{\mathrm{sess}}[r.\mathrm{session}] \ge 3$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $\textsc{Denylisted}(r)$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $\neg\textsc{Take}(r.\mathrm{address}, r.\mathrm{/24}, r.\mathrm{ASN})$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $p = \mathrm{web}$ \textbf{and} ($\mathrm{score}(r) < \theta_w$ \textbf{or} $\mathrm{origin}(r) \notin \mathcal{O}$) \textbf{then} \Return \textsc{Refuse}$()$
\State $(c, \mathrm{live}, v) \gets \textsc{CheckNumber}(r.\mathrm{number})$
\State \textbf{if} $\neg\mathrm{live}$ \textbf{or} $c \in \{\text{off-list}, \text{premium}\}$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $\neg\textsc{ValidBody}(r)$ \textbf{or} $r.\mathrm{number} \in \mathcal{X}$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $(r.\mathrm{bulk} \lor r.\mathrm{test}) \land \neg r.\mathrm{credential}$ \textbf{then} \Return \textsc{Refuse}$()$
\State $s \gets$ \scoreeq;\ $t \gets \textsc{Tier}(s, v, r.\mathrm{history})$
\State \textbf{if} $t = \mathrm{block}$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $t = \mathrm{challenge}$ \textbf{then} \Return \textsc{Challenge}$()$
\State \textbf{if} $\neg\textsc{Take}(\mathrm{backoff}[r.\mathrm{number}])$ \textbf{then} \Return \textsc{Refuse}$()$
\State \textbf{if} $\neg\textsc{Take}(m \cdot K[r.\mathrm{source}, p, \mathrm{country}(r.\mathrm{number})])$ \textbf{then} \Return \textsc{Refuse}$()$
\State $\mathrm{mode} \gets \textsc{Mode}(\mathrm{budget})$;\ \textbf{if} $\mathrm{mode} = \mathrm{emergency}$ \textbf{then} \Return \textsc{Refuse}$()$
\State $(\mathrm{ok}, g) \gets \textsc{Reserve}(\mathrm{count}, \mathrm{spend}, \mathrm{slot}[r.\mathrm{block}])$
\State \textbf{if} $\neg\mathrm{ok}$ \textbf{then} \Return \textsc{Refuse}$()$
\State $t \gets \textsc{Raise}(t, g, r.\mathrm{history})$;\ \textbf{if} $t = \mathrm{challenge}$ \textbf{then} \Return \textsc{Challenge}$()$
\State \textsc{Dispatch}$(r, t)$;\ \textsc{Write}$(S)$
\State \Return \textsc{Reply}$()$
\EndProcedure
\Statex
\Procedure{OnOutcome}{$x$, $o$}
\ForAll{$k \in \mathrm{keys}(x)$}
  \State $n_k \gets n_k + 1$;\ $v_k \gets v_k + [o = \mathrm{verified}]$
  \State \textbf{if} $n_k \ge 50$ \textbf{and} $v_k < 0.1\, n_k$ \textbf{then} \textsc{Denylist}$(k, 24\,\mathrm{h})$
\EndFor
\State \textbf{if} $o = \mathrm{verified}$ \textbf{then} $\mathrm{trusted}[x.\mathrm{number}] \gets \mathrm{true}$
\State $b \gets x.\mathrm{block}$;\ $S_c[b] \gets \max(-ch,\, S_c[b] + \delta_c(o))$
\State \textbf{if} $o = \mathrm{verified}$ \textbf{then} $S_s[b] \gets \max(-ch,\, S_s[b] + \delta_s(x))$
\State \textbf{if} $S_c[b] > h$ \textbf{or} $S_s[b] > h$ \textbf{then} \textsc{Verdict}$(b, 1\,\mathrm{h})$;\ $S_c[b], S_s[b] \gets 0$
\EndProcedure
\end{algorithmic}
