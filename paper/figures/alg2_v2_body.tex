% The v2 pipeline (Section III-C) and its feedback loop (Section III-D), as implemented in src/otp_guard/pipeline.py
% and feedback.py (release 2.9.0), in plain words. Shared by main.tex (Algorithm 2) and the standalone alg2_v2.tex for Word.
\begin{algorithmic}[1]
\Procedure{ProcessRequest}{$r$} \Comment{a refusal releases every claim taken so far}
\State establish the platform from the app attestation; refuse a proxy address or an invalid attestation \Comment{Step 0}
\State refuse unless $r$ carries a valid signed session bound to its fingerprint, with at most 3 requests in 10 min \Comment{Step 1}
\State refuse a denylisted fingerprint or address; take one slot each for the address, its /24 and its ASN \Comment{Step 2}
\State \textbf{if} $r$ is a web request \textbf{then} refuse a low reCAPTCHA score or a foreign origin \Comment{Steps 3--4}
\State check the number before paying for it: refuse a country off the allow-list, a premium prefix or a dead number; note sequential bursts and VoIP; read the verdict stage $v$ of its block \Comment{Step 5}
\State refuse an invalid body or an excluded number; the bulk and test flags need an internal credential \Comment{Step 6}
\State $s \gets$ \scoreeq; $t \gets \textsc{Tier}(s)$; for a client without verified history, raise $t$ to challenge if $v = 1$ and to downgrade if $v = 2$ \Comment{Step 7}
\State \textbf{if} $t$ = block \textbf{then refuse}; \textbf{if} $t$ = challenge \textbf{then return} the challenge \Comment{a solved one scores $-20$ on retry}
\State take the number's backoff slot: 60\,s doubling to 1\,h, five sends a day \Comment{Step 8}
\State take the source caps, scaled by the adaptive multiplier \Comment{Step 9}
\State set the operating mode from the hourly budget: elevated above 80\%, emergency at 100\% \Comment{Step 10}
\State reserve the hourly count, the spend and the block's counter slot in one operation; past the quota, raise $t$ for a client without verified history \Comment{Step 11}
\State send by SMS, or by push or WhatsApp if $t$ = downgrade, after a wait if $t$ = delay; write the state
\State \Return one uniform body after at least 400\,ms \Comment{the same body for sent, dropped and refused}
\EndProcedure
\Statex
\Procedure{OnOutcome}{send, outcome} \Comment{verified, failed or undelivered; applied once per transition}
\State count the outcome on every key of the send: address, /24, ASN, fingerprint, session, country, prefix and number
\State denylist a fine key for 24\,h once it has 50 resolved sends and fewer than 10\% verified
\State \textbf{if} verified \textbf{then} mark the number trusted \Comment{verified history exempts a client from both destination policies}
\State update the block's conversion statistic, and on a verification its speed statistic, each floored at $-ch$
\State \textbf{if} either statistic crosses $h$ \textbf{then} issue a verdict for 1\,h (stage 1, or stage 2 if one is active) and reset both
\EndProcedure
\end{algorithmic}
